\usepackage{booktabs}
\usepackage{float}
\usepackage{afterpage}

% The bibliography style (plainnat, set via biblio-style) prints DOIs with a
% \doi{...} command. The doi package defines it to typeset the DOI as a clickable
% link and, crucially, reads its argument with URL catcodes so special characters
% in DOIs (e.g. the underscore in 10.1007/978-3-662-43550-2_5) do not break the
% build the way an ordinary macro argument would.
\usepackage{doi}

% plainnat is used for the DOIs above, but natbib also derives citation
% punctuation from the style *name*: \bibstyle@plainnat (natbib.sty) presets
% square brackets and a comma between multiple citations, which is why the PDF
% read [Chang et al., 2015] while the HTML build (pandoc citeproc, which ignores
% biblio-style) read (Chang et al. 2015). Restore the parenthetical punctuation
% explicitly. The six arguments are: opening bracket, closing bracket, separator
% between multiple citations, style letter (a = author-year), separator between
% author and year, and separator before extra text after the year. These values
% are natbib's own defaults, i.e. what apalike used before the switch.
%
% This must come after \bibliographystyle{plainnat}, which pandoc emits just
% before it includes this file; \bibpunct assigns globally, so it wins.
\bibpunct{(}{)}{;}{a}{,}{,}

% Back-of-book index. \index{...} markers are placed inline in the chapter
% source (dropped silently in the HTML build, typeset by makeindex in the PDF).
% The index itself is emitted by \printindex in after_body.tex.
\usepackage{makeidx}
\makeindex

% The book class typesets the index with \twocolumn, whose output routine fills
% the first column to the full text height and leaves the final page unbalanced;
% combined with this book's strong widow/club penalties (set below), the first
% column can run past the bottom margin. Retypeset the index with multicol, which
% balances the two columns, keeps them inside the text block, and breaks to a new
% page cleanly. The penalties are also reset locally so entries break freely.
\usepackage{multicol}
\makeatletter
\renewenvironment{theindex}{%
  \columnseprule \z@
  \columnsep 35\p@
  \chapter*{\indexname}%
  \markboth{\indexname}{\indexname}%
  \thispagestyle{plain}%
  \parindent\z@
  \parskip\z@ \@plus .3\p@\relax
  \let\item\@idxitem
  \raggedright
  \widowpenalty\z@ \clubpenalty\z@ \clubpenalties 1 \z@ \interlinepenalty\z@
  \begin{multicols}{2}%
}{%
  \end{multicols}%
}
\makeatother

% Suppress the default pandoc/LaTeX title page. pandoc emits \maketitle from the
% YAML title/author/date, which produces a redundant title page (stamped with the
% build date) in front of our own front matter. The real title, copyright, and
% dedication pages are supplied by frontpage.tex (generated from index.rmd and
% included via before_body), so make \maketitle a no-op here.
\renewcommand{\maketitle}{}

% Advance to the next recto (odd) page, leaving any skipped verso completely
% blank -- no header, no page number. Used in the generated front matter
% (frontpage.tex) to put the title page and dedication on rectos, with the
% copyright page on the verso of the title page.
\newcommand{\clearrecto}{\clearpage\ifodd\value{page}\else\thispagestyle{empty}\null\clearpage\fi}

% Define book size
% margin sets the base 1in margin on all sides; bindingoffset adds a 0.25in
% gutter to the inner (spine) edge, mirrored on recto/verso, so the inside
% margin is 1.25in and the outside 1in -- the conventional allowance for
% perfect binding.
\usepackage[paperwidth=7in, paperheight=10in, margin=1in, bindingoffset=0.25in]{geometry}

% Running headers.
% The default book class typesets running heads in ALL CAPS at full size. On
% this narrow (~5in) text block, long section titles then run into the page
% number (e.g. "...DISCRETE CHARACTERS101"). Use smaller, mixed-case running
% heads so long titles fit, with the page number on the outer edge and the
% running title on the inner edge (further from the page number).
\usepackage{fancyhdr}
\setlength{\headheight}{14pt}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE,RO]{\small\thepage}
\fancyhead[RE]{\small\nouppercase{\leftmark}}
\fancyhead[LO]{\small\nouppercase{\rightmark}}
\renewcommand{\headrulewidth}{0.4pt}
% Chapter-opening pages use the "plain" style; put their page number in the foot.
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\small\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% Truly-blank versos. The book class is openright, so every \chapter -- and
% \tableofcontents -- issues \cleardoublepage, which inserts a blank verso
% whenever the preceding matter ends on a recto. The default \cleardoublepage
% leaves that inserted page in the active (fancy) page style, so it keeps a
% running head and folio. Redefine it to stamp the skipped verso empty: no
% header, no page number (the page is still counted, just not printed). This
% governs both the front-matter versos and every chapter-ending verso.
\makeatletter
\renewcommand{\cleardoublepage}{%
  \clearpage
  \ifodd\c@page\else
    \null\thispagestyle{empty}\newpage
  \fi}
\makeatother

% Widow and orphan control.
% A "widow" is the last line of a paragraph stranded at the top of a page; an
% "orphan" (club line) is the first line of a paragraph stranded at the bottom.
% 10000 is TeX's "infinite" penalty, telling it to avoid such breaks wherever it
% can. We also guard the lines around displayed equations, of which this book has
% many.
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
\postdisplaypenalty=10000

% Keep section headings from being stranded near the foot of a page with only a
% line or two of their text following. \clubpenalties makes the penalty apply to
% the first *two* lines after a break (i.e. after a heading), not just the first,
% so a heading pulls at least a couple of lines with it to the next page.
\clubpenalties 3 10000 10000 0

% The book class uses \flushbottom (every page flush to the bottom margin), which
% leaves TeX no vertical slack to honour the penalties above -- it will still
% break rather than under-fill a page. \raggedbottom lets pages end a line or two
% short so the penalties can actually take effect. The trade-off is that facing
% pages may not always end at exactly the same height.
\raggedbottom
